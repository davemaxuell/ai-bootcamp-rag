# BUFS Multilingual Admissions RAG — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Streamlit RAG app that answers foreign students' BUFS admissions questions in any language with cited answers, plus a personalized required-documents checklist.

**Architecture:** PDFs are extracted to metadata-tagged chunks, embedded with OpenAI and stored in a persistent Chroma vector DB. At runtime a retriever pulls top-k chunks; an answerer generates a grounded, cited reply in the user's language and refuses when context is insufficient; a checklist builder produces a per-program document list. A Streamlit UI (`app.py`) exposes Chat and Checklist tabs.

**Tech Stack:** Python 3 (`py -3` launcher, pythoncore-3.14), LangChain, ChromaDB, OpenAI (`text-embedding-3-small`, `gpt-4o-mini`), pdfplumber + pypdf, Streamlit, pytest.

## Global Constraints

- Python invoked as `py -3` on Windows (the interpreter that holds installed packages — NOT the msys2 `python`).
- API key loaded from `.env` (`OPENAI_API_KEY`) via `python-dotenv`; never hardcode or commit it.
- Embedding model: `text-embedding-3-small`. Chat model: `gpt-4o-mini`. Collection name: `bufs_admissions`.
- Every factual answer must cite `source` file + `page`; on insufficient/empty retrieval the answerer must refuse WITHOUT calling the LLM.
- Deployable app is `app.py` (Streamlit cannot run `.ipynb`). Notebooks are demo/exploration only.
- Tests that need no API (extraction, chunking, metadata, refusal guard, citation format, checklist assembly with stubbed retrieval) are the TDD unit tests. API-dependent behavior (embedding, generation) is covered by a smoke script + gold eval, run manually.

```python
from dataclasses import dataclass
from pathlib import Path
import pdfplumber
from pypdf import PdfReader
from config import PDF_DIR, meta_for


@dataclass
class Chunk:
    text: str
    source: str
    page: int
    lang: str
    level: str


def split_text(text: str, size: int = 2000) -> list[str]:
    text = text.strip()
    if not text:
        return []
    return [text[i:i + size] for i in range(0, len(text), size)]


def _page_texts(pdf_path) -> list[str]:
    """Per-page text: pdfplumber first (better tables), pypdf fallback."""
    out = []
    with pdfplumber.open(pdf_path) as pdf:
        plumber_pages = [(p.extract_text() or "") for p in pdf.pages]
    reader = PdfReader(str(pdf_path))
    for i, ptext in enumerate(plumber_pages):
        if len(ptext.strip()) < 20 and i < len(reader.pages):
            ptext = reader.pages[i].extract_text() or ptext
        out.append(ptext)
    return out


def extract_chunks(pdf_path) -> list[Chunk]:
    fname = Path(pdf_path).name
    m = meta_for(fname)
    chunks = []
    for page_no, ptext in enumerate(_page_texts(pdf_path), start=1):
        for piece in split_text(ptext):
            chunks.append(Chunk(piece, fname, page_no, m["lang"], m["level"]))
    return chunks


def extract_all() -> list[Chunk]:
    paths = sorted(PDF_DIR.glob("*.pdf")) + sorted(PDF_DIR.glob("*.PDF"))
    chunks = []
    for p in paths:
        chunks.extend(extract_chunks(p))
    return chunks
```

## Task 3: Build persistent Chroma store (API)

**Files:**
- Create: `ingest.py`

**Interfaces:**
- Consumes: `extract.extract_all`, `config.CHROMA_DIR`, `config.COLLECTION`, `config.EMBED_MODEL`.
- Produces: `ingest.build()` which writes a persistent Chroma collection to `CHROMA_DIR`; `ingest.get_vectorstore()` returning a loaded `langchain_chroma.Chroma` for reuse by the retriever.

- [ ] **Step 1: Write `ingest.py`**

```python
from dotenv import load_dotenv
from langchain_chroma import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain_core.documents import Document
from config import CHROMA_DIR, COLLECTION, EMBED_MODEL
from extract import extract_all

load_dotenv()


def _embeddings():
    return OpenAIEmbeddings(model=EMBED_MODEL)


def get_vectorstore() -> Chroma:
    return Chroma(
        collection_name=COLLECTION,
        embedding_function=_embeddings(),
        persist_directory=str(CHROMA_DIR),
    )


def build() -> int:
    chunks = extract_all()
    docs = [
        Document(
            page_content=c.text,
            metadata={"source": c.source, "page": c.page, "lang": c.lang, "level": c.level},
        )
        for c in chunks
    ]
    store = get_vectorstore()
    store.add_documents(docs)
    return len(docs)


if __name__ == "__main__":
    n = build()
    print(f"indexed {n} chunks into {CHROMA_DIR}")
```

- [ ] **Step 1: Write `retriever.py`**

```python
from ingest import get_vectorstore


def retrieve(query: str, k: int = 6, level: str | None = None):
    store = get_vectorstore()
    flt = None
    if level:
        flt = {"level": {"$in": [level, "general"]}}
    return store.similarity_search(query, k=k, filter=flt)
```

- [ ] **Step 3: Write `answerer.py`**

```python
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from config import CHAT_MODEL

load_dotenv()

REFUSAL = ("I could not find this in the official BUFS admissions documents. "
           "Please contact the BUFS admissions office to confirm.")

SYSTEM = (
    "You are a BUFS admissions assistant for foreign students. "
    "Answer ONLY from the provided context. "
    "Reply in the SAME language as the user's question. "
    "If the context does not contain the answer, say you don't know and refer them "
    "to the admissions office — never invent deadlines, fees, or requirements. "
    "Always end with a 'Sources:' line listing the file names and page numbers you used."
)


def format_context(docs) -> str:
    blocks = []
    for d in docs:
        m = d.metadata
        blocks.append(f"[source: {m['source']} | page: {m['page']}]\n{d.page_content}")
    return "\n\n---\n\n".join(blocks)


def answer(query: str, docs: list) -> dict:
    if not docs:
        return {"text": REFUSAL, "citations": []}
    llm = ChatOpenAI(model=CHAT_MODEL, temperature=0)
    ctx = format_context(docs)
    msg = llm.invoke([
        ("system", SYSTEM),
        ("human", f"Context:\n{ctx}\n\nQuestion: {query}"),
    ])
    citations = [{"source": d.metadata["source"], "page": d.metadata["page"]} for d in docs]
    return {"text": msg.content, "citations": citations}
```

```python
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from config import CHAT_MODEL
from answerer import format_context, REFUSAL
from retriever import retrieve

load_dotenv()

SYSTEM = (
    "You are a BUFS admissions assistant. Using ONLY the context, produce a clear "
    "checklist of required application documents and their deadlines for the given "
    "applicant. Reply in the requested language. Use a numbered list. If the context "
    "lacks the information, say so and refer them to the admissions office. "
    "End with a 'Sources:' line of file names and page numbers."
)


def build_checklist(level: str, program: str, applicant_type: str,
                    lang: str, retriever_fn=retrieve) -> dict:
    query = f"required documents checklist deadlines {level} {program} {applicant_type}"
    docs = retriever_fn(query, k=8, level=level)
    if not docs:
        return {"text": REFUSAL, "citations": []}
    llm = ChatOpenAI(model=CHAT_MODEL, temperature=0)
    ctx = format_context(docs)
    ask = (f"Applicant: level={level}, program={program}, type={applicant_type}. "
           f"Answer language: {lang}.")
    msg = llm.invoke([("system", SYSTEM), ("human", f"Context:\n{ctx}\n\n{ask}")])
    citations = [{"source": d.metadata["source"], "page": d.metadata["page"]} for d in docs]
    return {"text": msg.content, "citations": citations}
```

```python
import streamlit as st
from retriever import retrieve
from answerer import answer
from checklist import build_checklist

st.set_page_config(page_title="BUFS Admissions Assistant", page_icon="🎓")
st.title("🎓 BUFS Admissions Assistant")
st.caption("Ask in any language. Answers come only from official BUFS admission documents.")

chat_tab, list_tab = st.tabs(["Ask a question", "My document checklist"])

with chat_tab:
    q = st.text_input("Your question (any language)")
    if st.button("Ask") and q:
        with st.spinner("Searching official documents..."):
            out = answer(q, retrieve(q))
        st.write(out["text"])
        if out["citations"]:
            st.caption("Sources: " + ", ".join(
                f"{c['source']} p.{c['page']}" for c in out["citations"]))

with list_tab:
    level = st.selectbox("Program level", ["undergraduate", "graduate"])
    program = st.text_input("Program / major", "Korean Language")
    applicant_type = st.selectbox("Applicant type", ["new", "transfer"])
    lang = st.text_input("Answer language", "English")
    if st.button("Build my checklist"):
        with st.spinner("Building your checklist..."):
            out = build_checklist(level, program, applicant_type, lang)
        st.write(out["text"])
        if out["citations"]:
            st.caption("Sources: " + ", ".join(
                f"{c['source']} p.{c['page']}" for c in out["citations"]))
```

```python
import sys
from pathlib import Path
import yaml

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
from retriever import retrieve
from answerer import answer

def main():
    items = yaml.safe_load((Path(__file__).parent / "gold_questions.yaml").read_text(encoding="utf-8"))
    passed = 0
    for it in items:
        docs = retrieve(it["q"])
        out = answer(it["q"], docs)
        sources = " ".join(c["source"] for c in out["citations"])
        ok = it["expect_source_contains"] in sources and out["text"].strip() != ""
        print(f"[{'PASS' if ok else 'FAIL'}] {it['q'][:30]} -> {sources[:60]}")
        passed += ok
    print(f"{passed}/{len(items)} passed")

if __name__ == "__main__":
    main()
```
